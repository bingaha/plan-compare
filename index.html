Agent 套餐价格对比

导入数据
全局用量混合比 综合单价 = 输入占比 ×（命中率 × 缓存价 + (1−命中率) × 输入价）+ 输出占比 × 输出价 列颜色：名称 / 可维护数据 / 计算生成；数据在「管理数据」里维护 
导入 JSON 格式（与「导出」按钮导出的整份数据一致；字段严格校验——多余字段、缺失字段、错误类型、非法取值都会被拒绝，不会导入）
根节点必须是对象，不能是数组；一行一行的扁平行数据（如旧版格式、表格导出）无法导入，需要先转换成下面的结构。

{
  "kind": "plan-compare-data",               // 可省略；写了就必须是这个值
  "format": 2,                               // 可省略；写了就必须是 2
  "exportedAt": "2026-09-18T09:55:35.264Z",  // 可省略；ISO 时间字符串
  "global": {                                // 必填；全局用量混合比
    "inputRatio": 0.99,                      // 输入占比，0~1 的数字
    "cacheHitRate": 0.95,                    // 缓存命中率，0~1 的数字
    "outputRatio": 0.01                      // 输出占比，0~1 的数字；inputRatio + outputRatio 应≈1
  },
  "platforms": [                             // 必填；平台数组
    {
      "id": "p-zai",                         // 非空字符串，全局唯一
      "name": "智谱 Z.ai",                   // 非空字符串
      "origin": "builtin",                   // 可省略；"builtin" | "custom"
      "models": [                            // 必填（可为空数组）；该平台各模型的单价，每 Mtok，与套餐额度同币种
        {
          "id": "m-53",                      // 非空字符串，全局唯一
          "name": "GLM-5.3",                 // 非空字符串；评分列按模型名匹配
          "priceIn": 690, "priceOut": 2400, "priceCache": 170,   // 各 ≥0 的数字或 null：输入价 / 输出价 / 缓存命中价
          "quota": 128571,                   // 可省略；>0 的数字或 null：该模型自有额度（仅额度制）。不填则用套餐 quota 作默认值
          "note": "抵扣系数等说明",           // 可省略；字符串
          "origin": "builtin"                // 可省略；"builtin" | "custom"
        }
      ],
      "plans": [                             // 必填（可为空数组）；该平台的套餐
        {
          "id": "pl-lite",                   // 非空字符串，全局唯一
          "name": "V3 Lite",                 // 非空字符串
          "price": 283.2,                    // ≥0 的数字或 null；人民币实付价
          "period": "quarter",               // "month" | "quarter" | "year"
          "mode": "quota",                   // "quota" 额度制 | "usage" 用量制
          "quota": 128571,                   // mode=quota 用；>0 的数字或 null（用量制可省略）；套餐默认额度，按计价币种。模型单独填了 quota 时优先用模型的
          "usageM": { "m-53": 1000 },        // mode=usage 用；键必须是本平台的模型 id，值是 >0 的数字或 null
          "consumeRate": 2,                  // 可省略；消耗倍率，>0 的数字或 null，缺省按 1
          "priceRate": 1,                    // 可省略；价格倍率，>0 的数字或 null，缺省按 1
          "note": "备注",                    // 可省略；字符串
          "origin": "builtin"                // 可省略；"builtin" | "custom"
        }
      ]
    }
  ],
  "scores": [                                // 必填；评分按模型名对应到主表列
    { "src": "Artificial Analysis", "model": "GLM-5.3", "value": 45 }
  ]
}

主表每行 = 平台下的 套餐 × 模型 自动配对，价格与倍率在套餐/模型层只存一份。额度制下每行有效额度 = 模型自有额度 ?? 套餐额度（模型没填就用套餐的）。导入与现有数据合并：global 取导入值；平台按 id、评分按「评分源+模型名」，同者更新、新者追加，现有其他内容保留，并标记为「已修改」。
管理数据改动实时保存到浏览器本地，并同步到主表
    导入 JSON严格校验后与现有数据合并（同 id 更新、新条目追加），不会清掉旧数据
